                FloatingIcon.Visible = false

-- Function to create tab
local function createTab(name, icon, setupContent)
    local tabBtn = Instance.new("TextButton")
    tabBtn.Size = UDim2.new(0, 38, 0, 38)
    tabBtn.BackgroundTransparency = 1
    tabBtn.Text = icon
    tabBtn.TextColor3 = CONFIG.TextColor
    tabBtn.Font = Enum.Font.GothamBold
    tabBtn.TextSize = 18
    tabBtn.Parent = TabsContainer
    
    tabBtn.MouseEnter:Connect(function()
        TweenService:Create(tabBtn, TweenInfo.new(0.2), {TextColor3 = CONFIG.AccentColor}):Play()
    end)
    
    tabBtn.MouseLeave:Connect(function()
        if activeTab ~= name then
            TweenService:Create(tabBtn, TweenInfo.new(0.2), {TextColor3 = CONFIG.TextColor}):Play()
        end
    end)
    
    tabBtn.MouseButton1Click:Connect(function()
        playClick()
        -- Reset all tabs
        for _, child in pairs(TabsContainer:GetChildren()) do
            if child:IsA("TextButton") then
                TweenService:Create(child, TweenInfo.new(0.2), {TextColor3 = CONFIG.TextColor}):Play()
            end
        end
        TweenService:Create(tabBtn, TweenInfo.new(0.2), {TextColor3 = CONFIG.AccentColor}):Play()
        
        activeTab = name
        
        -- Move indicator
        TabIndicator.Visible = true
        TweenService:Create(TabIndicator, TweenInfo.new(0.25, Enum.EasingStyle.Quad), {
            Size = UDim2.new(0, tabBtn.AbsoluteSize.X, 0, 3),
            Position = UDim2.new(0, tabBtn.AbsolutePosition.X - MainFrame.AbsolutePosition.X, 1, -3)
        }):Play()
        
        -- Clear and setup content
        for _, child in pairs(TabContent:GetChildren()) do
            child:Destroy()
        end
        
        TabOverlay.Visible = true
        setupContent(TabContent)
    end)
    
    return tabBtn
end
